the rest load lazily. */ -}} {{- $priority := cond (eq .Ordinal 0) (dict "fetchpriority" "high") (dict "loading" "lazy") -}} {{- $subtype := "" -}} {{- with $res }}{{ $subtype = .MediaType.SubType }}{{ end -}} {{- if in (slice "png" "jpeg" "webp" "tiff" "bmp" "gif") $subtype -}} {{- $attributes = merge $attributes $priority (dict "decoding" "async") -}} {{- if eq $subtype "gif" -}} {{- /* Possibly animated: never re-encoded. */ -}} {{- $attributes = merge $attributes (dict "width" (string $res.Width) "height" (string $res.Height)) -}} {{- else if gt $res.Width 650 -}} {{- $small := $res.Process "resize 650x webp q80" -}} {{- $large := $res.Process (printf "resize %dx webp q80" (int (math.Min 1300 $res.Width))) -}} {{- $attributes = merge $attributes (dict "src" $small.RelPermalink "srcset" (printf "%s %dw, %s %dw" $small.RelPermalink $small.Width $large.RelPermalink $large.Width) "sizes" "(min-width: 700px) 650px, calc(100vw - 3rem)" "width" (string $small.Width) "height" (string $small.Height)) -}} {{- else -}} {{- /* Small enough to keep its size, but still served as WebP. */ -}} {{- $image := $res -}} {{- if ne $subtype "webp" }}{{ $image = $res.Process "webp q80" }}{{ end -}} {{- $attributes = merge $attributes (dict "src" $image.RelPermalink "width" (string $image.Width) "height" (string $image.Height)) -}} {{- end -}} {{- else if eq $subtype "svg" -}} {{- /* Hugo cannot measure an SVG, so its viewBox supplies the width and height attributes.
